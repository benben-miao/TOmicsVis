Installation

Get started in minutes

# Install from CRAN
install.packages("TOmicsVis")

# Load the package
library(TOmicsVis)
# Install from GitHub
install.packages("devtools")
devtools::install_github("benben-miao/TOmicsVis")

# Load the package
library(TOmicsVis)
# Install required Bioconductor packages
install.packages("BiocManager")
BiocManager::install(c(
  "ComplexHeatmap", 
  "EnhancedVolcano", 
  "clusterProfiler", 
  "enrichplot", 
  "impute", 
  "preprocessCore", 
  "Mfuzz"
))

Quick Start

# Load example data
data(gene_expression)
data(samples_groups)

# Create a volcano plot
data(degs_stats)
volcano_plot(degs_stats)

# Launch Shiny app
TOmicsVis::tomicsvis()
